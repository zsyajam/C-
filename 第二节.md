# 循环语句

## 循环

~~~ c
#include "stdio.h"

int i=5;
int main()
{
    while(i--)
    {
        printf("%d\n",i);
        
    }
    return 0;
}


~~~

~~~ c
#include "stdio.h"

int i=5;
int main()
{
    while(i>3)
    {
        printf("%d\n",i);
        
    }
    return 0;
}
~~~

~~~ c
#include "stdio.h"

int i=0;
int main()
{
    do
    {
        printf("%d\n",i);
        i++;
        
    }	while(i==5);
    
    printf("循环执行完了")
        
        
    return 0;
}
~~~

~~~ C
#include "stdio.h"

int num;
int i;
int main()
{
	printf("请输入一个整数:");
    scanf("%d",&num);
    
    do
    {
    	num=num/10;        //num/=10;
        i++;  
    }while(num>0)
    return 0;
}
~~~

~~~ c
#include "stdio.h"

int i;
int main()
{
   for(i=5;i>0;i--)
   {
   printf("%d/n",i);
   if(i==2)
   break;
   }
    return 0;
}
~~~
